  function activeUser() { return signedIn() && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.disabled == false; }
  function own(uid) { return activeUser() && request.auth.uid == uid; }
  function limited() { return request.query.limit != null && request.query.limit <= 100; }
  function symbols() { return ['BTC-USD','ETH-USD','XRP-USD','SOL-USD','BNB-USDT','DOGE-USD','TON-USDT','ADA-USD','EUR-USD','GBP-USD','XAU-USD','XAG-USD','SPX-USD','NDX-USD']; }
  function preferences(p) {
   return p is map && p.keys().hasOnly(['hideBalance','notifications','reduceMotion','notificationSounds','opportunitySound','securitySound','vibration','marketingNotifications','chartLocale','theme','fontSize','numberFormat','referenceCurrency','timezone','chartPair','chartInterval','chartStyle','grid','crosshair','priceLine','saveDrawings','indicators','favorites','notificationEvents','notificationChannels']) && p.size()==25 &&
    p.hideBalance is bool && p.notifications is bool && p.reduceMotion is bool && p.notificationSounds is bool && p.opportunitySound is bool && p.securitySound is bool && p.vibration is bool && p.marketingNotifications is bool &&
    p.chartLocale in ['ar','en'] && p.theme in ['light','dark','system'] && p.fontSize in ['small','normal','large'] && p.numberFormat in ['latin','arabic'] && p.referenceCurrency in ['USD','EUR','GBP'] && p.timezone is string && p.timezone.size()>0 && p.timezone.size()<=80 &&
    p.chartPair in symbols() && p.chartInterval in ['1','5','15','60','240','D'] && p.chartStyle in ['1','2','3'] && p.grid is bool && p.crosshair == true && p.priceLine == true && p.saveDrawings == false &&
    p.indicators is list && p.indicators.size()<=5 && p.indicators.hasOnly(['MA','EMA','RSI','MACD','BB']) && p.favorites is list && p.favorites.size()<=14 && p.favorites.hasOnly(symbols()) &&
    p.notificationChannels == {'push':false,'email':false,'sms':false} &&
    p.notificationEvents is map && p.notificationEvents.size()==12 && p.notificationEvents.keys().hasOnly(['opportunity','contract_started','contract_ended','settlement','deposit','withdrawal','referral','agent','support','security','announcement','reminder']) &&
    p.notificationEvents.security == true && p.notificationEvents.opportunity is bool && p.notificationEvents.contract_started is bool && p.notificationEvents.contract_ended is bool && p.notificationEvents.settlement is bool && p.notificationEvents.deposit is bool && p.notificationEvents.withdrawal is bool && p.notificationEvents.referral is bool && p.notificationEvents.agent is bool && p.notificationEvents.support is bool && p.notificationEvents.announcement is bool && p.notificationEvents.reminder is bool;
  }
  function validProfile(p) { return p is map && p.keys().hasOnly(['displayName','phone','country','closureStatus']) && p.size()==4 && p.displayName is string && p.displayName.size()>0 && p.displayName.size()<=100 && p.phone is string && (p.phone == '' || p.phone.matches('\\+?[0-9 ()-]{6,24}')) && p.country is string && (p.country == '' || p.country.matches('[A-Z]{2}')) && p.closureStatus in ['open','requested']; }
  function sparkProfile(d,uid) { return d.keys().hasOnly(['uid','profile','preferences','tier','acceptedAt','revision','createdAt','updatedAt']) && d.size()==8 && d.uid==uid && validProfile(d.profile) && preferences(d.preferences) && d.tier in ['standard','advanced'] && (d.acceptedAt == null || d.acceptedAt is timestamp) && d.revision is int && d.revision>=0 && d.createdAt is timestamp && d.updatedAt==request.time; }
  function closureAllowed(uid,d) { return d.profile.closureStatus != 'requested' || !exists(/databases/$(database)/documents/orinAccounts/$(uid)) || get(/databases/$(database)/documents/orinAccounts/$(uid)).data.reservedCents==0; }
  match /sparkProfiles/{uid} {
   allow get: if own(uid) || (activeUser() && admin());
   allow list: if activeUser() && admin() && limited();
   allow create: if own(uid) && sparkProfile(request.resource.data,uid) && request.resource.data.tier=='standard' && request.resource.data.acceptedAt==null && request.resource.data.revision==0 && request.resource.data.createdAt==request.time && request.resource.data.profile.closureStatus=='open';
   allow update: if own(uid) && sparkProfile(request.resource.data,uid) && request.resource.data.createdAt==resource.data.createdAt && request.resource.data.revision==resource.data.revision+1 && (request.resource.data.tier==resource.data.tier || admin()) &&
    (request.resource.data.acceptedAt==resource.data.acceptedAt || resource.data.acceptedAt==null && request.resource.data.acceptedAt==request.time) &&
    (resource.data.profile.closureStatus!='requested' || request.resource.data.profile.closureStatus=='requested') && closureAllowed(uid,request.resource.data);
   allow delete: if false;
   match /inbox/{cid} {
    allow read: if own(uid);
    allow create: if activeUser() && admin() && request.resource.data.keys().hasOnly(['campaignId','createdAt','readAt']) && request.resource.data.campaignId==cid && request.resource.data.createdAt==request.time && request.resource.data.readAt==null &&
     getAfter(/databases/$(database)/documents/sparkCampaigns/$(cid)).data.status=='sent' && getAfter(/databases/$(database)/documents/sparkCampaigns/$(cid)).data.sentAt==request.time && uid in getAfter(/databases/$(database)/documents/sparkCampaigns/$(cid)).data.recipientIds;
    allow update: if own(uid) && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['readAt']) && request.resource.data.readAt==request.time;
    allow delete: if false;
   }
  }
  function marketConfig(c) { return c is map && c.keys().hasOnly(['defaultSymbol','enabledSymbols','defaultTimeframe','enabledTimeframes','defaultChartType','enabledChartTypes','allowedIndicators','allowedDrawingTools','volume','gridDefault','crosshairDefault','fullscreen','alerts','favorites']) && c.size()==14 &&
   c.enabledSymbols is list && c.enabledSymbols.size()>0 && c.enabledSymbols.size()<=14 && c.enabledSymbols.hasOnly(symbols()) && c.defaultSymbol in c.enabledSymbols &&
   c.enabledTimeframes is list && c.enabledTimeframes.size()>0 && c.enabledTimeframes.size()<=6 && c.enabledTimeframes.hasOnly(['1','5','15','60','240','D']) && c.defaultTimeframe in c.enabledTimeframes &&
   c.enabledChartTypes is list && c.enabledChartTypes.size()>0 && c.enabledChartTypes.size()<=3 && c.enabledChartTypes.hasOnly(['1','2','3']) && c.defaultChartType in c.enabledChartTypes &&
   c.allowedIndicators is list && c.allowedIndicators.size()<=5 && c.allowedIndicators.hasOnly(['MA','EMA','RSI','MACD','BB']) && c.allowedDrawingTools==[] && c.volume is bool && c.gridDefault is bool && c.crosshairDefault==true && c.fullscreen is bool && c.alerts==false && c.favorites is bool; }
  match /sparkConfiguration/markets {
   allow get: if activeUser();
   allow create,update: if activeUser() && admin() && request.resource.data.keys().hasOnly(['configuration','revision','updatedAt']) && marketConfig(request.resource.data.configuration) && request.resource.data.updatedAt==request.time && request.resource.data.revision is int && request.resource.data.revision==(resource==null?1:resource.data.revision+1);
   allow delete: if false;
  }
  function validCampaign(d,cid) { return d.keys().hasOnly(['id','title','message','image','audience','language','deep_link','starts_at','expires_at','status','revision','createdAt','sentAt','recipient_count','recipientIds']) && d.size()==15 && d.id==cid && cid.matches('[A-Za-z0-9_-]{1,160}') &&
   d.title is string && d.title.size()>0 && d.title.size()<=80 && d.message is string && d.message.size()>0 && d.message.size()<=500 && d.image is string && (d.image=='' || d.image.matches('/brand/[A-Za-z0-9/_.-]+\\.(png|jpg|webp|svg)')) && !d.image.matches('.*\\.\\..*') &&
   d.audience is string && d.audience.size()<=12000 && d.language in ['ar','en','all'] && d.deep_link in ['/notifications','/settings/security','/invitations','/referral','/support'] && d.starts_at is int && d.expires_at is int && d.expires_at>d.starts_at && d.expires_at>request.time.toMillis() && d.expires_at<=request.time.toMillis()+7776000000 &&
   d.status in ['draft','sent','cancelled'] && d.revision is int && d.revision>=0 && d.createdAt is timestamp && (d.sentAt==null || d.sentAt is timestamp) && d.recipientIds is list && d.recipientIds.size()<=200 && d.recipientIds.size()==d.recipientIds.toSet().size() && d.recipient_count==d.recipientIds.size(); }
  match /sparkCampaigns/{cid} {
   allow get: if activeUser() && admin();
   allow list: if activeUser() && admin() && limited();
   allow create: if activeUser() && admin() && validCampaign(request.resource.data,cid) && request.resource.data.status=='draft' && request.resource.data.revision==0 && request.resource.data.createdAt==request.time && request.resource.data.sentAt==null && request.resource.data.recipientIds==[];
   allow update: if activeUser() && admin() && validCampaign(request.resource.data,cid) && resource.data.status=='draft' && request.resource.data.revision==resource.data.revision+1 && request.resource.data.createdAt==resource.data.createdAt && (
    (request.resource.data.status in ['draft','cancelled'] && request.resource.data.sentAt==null && request.resource.data.recipientIds==[]) ||
    (request.resource.data.status=='sent' && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['status','revision','sentAt','recipient_count','recipientIds']) && request.resource.data.sentAt==request.time && request.resource.data.starts_at<=request.time.toMillis() && request.resource.data.recipient_count>0)
   );
   allow delete: if false;
  }
  match /sparkAnnouncements/{cid} {

    getAfter(/databases/$(database)/documents/sparkCampaigns/$(cid)).data.status=='sent' && getAfter(/databases/$(database)/documents/sparkCampaigns/$(cid)).data.sentAt==request.time &&

   allow update,delete,list: if false;
  }
